Ingress proxy cache is enabled for this release.

1. Add the cache zone via the ingress-nginx Helm release (namespace ingress, release ingress-nginx):

   Merge charts/gxa/docs/ingress-nginx-values-gxa-cache.yaml into controller values, then helm upgrade.

   http-snippet line (must match zoneName {{ .Values.ingress.cache.zoneName }}):

{{ include "app.ingress.controllerHttpSnippet" . | indent 4 }}

   See charts/gxa/docs/ingress-cache-controller-snippet.example.yaml

2. Prefer disabling the pod sidecar cache when using ingress cache:

   nginx.cache.enabled: false

3. Purge ingress cache: delete files under the controller cache path on ingress pods, or restart ingress controller pods.

1. Get the application URL by running these commands:
{{- if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "app.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "app.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}
